# 本地书籍缓存分割方案

## 概述

本方案用于将本地书籍内容缓存分割成一章一个小文件，以减少阅读时的卡顿。通过预先将每章内容提取并保存为独立文件，实现快速加载和流畅阅读体验。

## 核心代码

### 1. BookHelp.kt - 缓存管理核心

```kotlin
package io.legado.app.help.book

import io.legado.app.data.appDb
import io.legado.app.data.entities.Book
import io.legado.app.data.entities.BookChapter
import io.legado.app.utils.FileUtils
import io.legado.app.utils.StringUtils
import io.legado.app.utils.externalFiles
import io.legado.app.utils.createFileIfNotExist
import io.legado.app.model.localBook.LocalBook
import splitties.init.appCtx
import java.io.File

/**
 * 书籍缓存帮助类
 * 负责管理本地书籍的章节内容缓存
 */
object BookHelp {
    
    // 缓存目录
    private val downloadDir: File = appCtx.externalFiles
    private const val cacheFolderName = "book_cache"
    
    val cachePath: String = FileUtils.getPath(downloadDir, cacheFolderName)

    /**
     * 保存章节内容到缓存文件
     * 每个章节保存为一个独立的 .nb 文件
     */
    fun saveText(
        book: Book,
        bookChapter: BookChapter,
        content: String
    ) {
        if (content.isEmpty()) return
        
        // 创建目录结构: {缓存目录}/{书籍文件夹}/{章节文件.nb}
        val file = FileUtils.createFileIfNotExist(
            downloadDir,
            cacheFolderName,
            book.getFolderName(),
            bookChapter.getFileName()  // 生成文件名: 00001-xxxxx.nb
        )
        file.writeText(content)
        
        // 如果启用字数统计，更新章节字数
        if (book.isOnLineTxt && AppConfig.tocCountWords) {
            val wordCount = StringUtils.wordCountFormat(content.length)
            bookChapter.wordCount = wordCount
            appDb.bookChapterDao.upWordCount(
                bookChapter.bookUrl, 
                bookChapter.url, 
                wordCount
            )
        }
    }

    /**
     * 从缓存读取章节内容
     * 优先读取缓存文件，如果不存在则从原书读取并缓存
     */
    fun getContent(book: Book, bookChapter: BookChapter): String? {
        val file = File(
            downloadDir,
            FileUtils.getPath(cacheFolderName, book.getFolderName(), bookChapter.getFileName())
        )
        
        // 1. 检查缓存文件是否存在
        if (file.exists()) {
            val content = file.readText()
            if (content.isEmpty()) return null
            return content
        }
        
        // 2. 如果是本地书籍，从原书文件读取
        if (book.isLocal) {
            val content = LocalBook.getContent(book, bookChapter)
            // 读取后自动保存到缓存目录，便于后续快速读取
            if (content != null && content != "null") {
                saveText(book, bookChapter, content)
            }
            return content
        }
        
        return null
    }

    /**
     * 检查章节是否已缓存
     */
    fun hasContent(book: Book, bookChapter: BookChapter): Boolean {
        if (book.isLocalTxt) return true
        return File(
            downloadDir,
            FileUtils.getPath(cacheFolderName, book.getFolderName(), bookChapter.getFileName())
        ).exists()
    }

    /**
     * 清除书籍缓存
     */
    fun clearCache(book: Book) {
        val filePath = FileUtils.getPath(downloadDir, cacheFolderName, book.getFolderName())
        FileUtils.delete(filePath)
    }
}
```

### 2. BookChapter.kt - 章节文件名生成

```kotlin
package io.legado.app.data.entities

import io.legado.app.utils.MD5Utils
import android.annotation.SuppressLint

data class BookChapter(
    var url: String = "",
    var title: String = "",
    var isVolume: Boolean = false,
    var bookUrl: String = "",
    var index: Int = 0,
    var start: Long? = null,    // 章节在原文件中的起始字节位置
    var end: Long? = null,      // 章节在原文件中的结束字节位置
    // ... 其他字段
) {
    
    @Transient
    var titleMD5: String? = null

    private fun ensureTitleMD5Init() {
        if (titleMD5 == null) {
            titleMD5 = MD5Utils.md5Encode16(title)
        }
    }

    /**
     * 生成章节缓存文件名
     * 格式: {序号5位}-{标题MD5前5位}.nb
     * 例如: 00001-a3f5c.nb
     */
    @SuppressLint("DefaultLocale")
    fun getFileName(suffix: String = "nb"): String {
        ensureTitleMD5Init()
        return String.format("%05d-%s.%s", index, titleMD5, suffix)
    }
}
```

### 4. TextFile.kt - 本地TXT书籍内容提取

```kotlin
package io.legado.app.model.localBook

import io.legado.app.data.entities.Book
import io.legado.app.data.entities.BookChapter
import io.legado.app.model.localBook.LocalBook
import kotlin.math.min

class TextFile(private var book: Book) {

    companion object {
        // 缓冲区大小: 8MB
        private const val txtBufferSize = 8 * 1024 * 1024
        
        @Synchronized
        fun getContent(book: Book, bookChapter: BookChapter): String {
            return TextFile(book).getContent(bookChapter)
        }
    }

    private var charset: Charset = book.fileCharset()
    private var txtBuffer: ByteArray? = null
    private var bufferStart = -1L
    private var bufferEnd = -1L

    /**
     * 根据章节起止位置从大文件中提取内容
     */
    fun getContent(chapter: BookChapter): String {
        val start = chapter.start ?: 0
        val end = chapter.end ?: 0
        
        // 1. 检查缓冲区是否包含所需内容
        if (txtBuffer == null || start > bufferEnd || end < bufferStart) {
            // 2. 需要重新加载缓冲区
            LocalBook.getBookInputStream(book).use { bis ->
                bufferStart = txtBufferSize * (start / txtBufferSize)
                txtBuffer = ByteArray(min(txtBufferSize, bis.available() - bufferStart.toInt()))
                bufferEnd = bufferStart + txtBuffer!!.size
                bis.skip(bufferStart)
                bis.read(txtBuffer)
            }
        }

        // 3. 提取章节内容
        val count = (end - start).toInt()
        val buffer = ByteArray(count)

        @Suppress("ConvertTwoComparisonsToRangeCheck")
        if (start < bufferEnd && end > bufferEnd || start < bufferStart && end > bufferStart) {
            // 章节内容跨越缓冲区边界，从文件直接读取
            LocalBook.getBookInputStream(book).use { bis ->
                bis.skip(start)
                bis.read(buffer)
            }
        } else {
            // 章节内容在缓冲区内，直接复制
            txtBuffer!!.copyInto(
                buffer,
                0,
                (start - bufferStart).toInt(),
                (end - bufferStart).toInt()
            )
        }

        return String(buffer, charset).replace(padRegex, "　　")
    }
}
```

### 5. FileUtils.kt - 文件操作辅助

```kotlin
package io.legado.app.utils

import java.io.File

object FileUtils {

    /**
     * 创建文件(如果不存在)并返回File对象
     */
    fun createFileIfNotExist(
        vararg dirs: String
    ): File {
        val file = getFile(*dirs)
        if (!file.parentFile.exists()) {
            file.parentFile.mkdirs()
        }
        if (!file.exists()) {
            file.createNewFile()
        }
        return file
    }

    /**
     * 获取文件路径
     */
    fun getPath(vararg dirs: String): String {
        return dirs.joinToString(File.separator)
    }

    /**
     * 获取File对象
     */
    fun File.getFile(vararg dirs: String): File {
        var file = this
        for (dir in dirs) {
            file = File(file, dir)
        }
        return file
    }

    /**
     * 删除文件或目录
     */
    fun delete(path: String) {
        val file = File(path)
        if (file.isDirectory) {
            file.deleteRecursively()
        } else {
            file.delete()
        }
    }
}
```

## 性能优化建议

1. **预加载**: 在阅读当前章节时，预先加载下一章到缓存
2. **LRU缓存**: 内存中保留最近访问的3个章节内容
3. **异步写入**: 缓存写入使用IO线程，不阻塞主线程
4. **增量解析**: 只解析需要的章节，避免全文件扫描

---

## 注意事项

1. **编码问题**: TXT文件可能有多种编码(UTF-8, GBK等)，需要自动检测
2. **BOM头**: UTF-8文件可能有BOM头，需要处理
3. **换行符**: Windows(`\r\n`)、Linux(`\n`)、Mac(`\r`)换行符不同
4. **章节边界**: 以字节位置为准，而非字符位置(中英文宽度不同)
